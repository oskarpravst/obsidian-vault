        Scanner sc = new Scanner(System.in);

        int stOtrok = sc.nextInt();
        int stBesed = sc.nextInt();

            System.out.println(otroci.vrni(ixIzpadlega));

            otroci.odstrani(ixIzpadlega);
            stOtrok--;
        }
    }
}

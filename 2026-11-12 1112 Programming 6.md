```java
public static void main(...) {
	int a = 42; // a[42]
	int[] t = {1, 2, 3}; // t[->] [1, 2, 3]
	int[] u = {4, 5, 6}; // u[->] [4, 5, 6]
	int[] v = f(a, t, u); // v[->] [7, 8]
	
	
	
}

public static int f(int aa, int[] tt, int[] uu) {
	// aa[42]
	// tt[->] [1, 2, 3]
	// uu[->] [4, 5, 6]
	aa = 66; // aa[66]
	tt[1] = 10; // [1, 10, 3]
	uu = new int[]{7, 8}; // uu[->][7, 8], ne velja vec uu[->] [4, 5, 6]
	return uu;
	
}
```

izpis tabele:

```java
public class IzpisTabele {
    public static void main(String[] args) {
        int[][] t = {
            {3, 4, 1, 9},
            {7, 5, 10, 6},
            {12, 1, 0, 8}
        };

        for (int i = 0; i < t.length; i++) {
            for (int j = 0; j < t[i].length; j++) {
                System.out.print(t[i][j] + " ");
            }
            System.out.println();
        }
        
        // ali

        for (int[] vrstica: t) {
            for (int element: vrstica) {
                System.out.print(element + " ");
            }
            System.out.println();
        }
    }
}
```

tabela stolpcev 

```java
import java.util.Scanner;
import java.util.Arrays;

public class Stolpci {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt(); // stevilo vrstic

        int[][] stopnice = new int[n][];

        for (int i = 0; i < n; i++) {
            int[] vrstica = new int[i + 1];
            for (int j = 0; j <= i; j++) {
                vrstica[j] = j + 1;
            }
            stopnice[i] = vrstica;
        }

        for (int[] vrstica: stopnice) {
            System.out.println(Arrays.toString(vrstica));
        }
    }
}
```
